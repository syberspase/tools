            analyzer = FinBERTSentimentAnalyzer(); \
            result = analyzer.analyze_text('Apple surges on record earnings'); \
            print(f'Score: {result[\"score\"]}')"

# Test signal generation
python3 -c "from finbert_trading_agent import FinBERTTradingAgent; \
            agent = FinBERTTradingAgent(); \
            ohlcv = agent.get_ohlcv_from_backend('AAPL', '15m'); \
            signal = agent.generate_trading_signal('AAPL', ohlcv); \
            print(f'Action: {signal[\"action\"]}, Confidence: {signal[\"confidence\"]:.2%}')"
